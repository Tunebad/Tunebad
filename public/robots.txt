# TuneBad robots policy.
#
# The site runs on free hosting tiers, where crawler traffic -- not people --
# is what actually uses up the quota (Oct 2026: ~99% of requests were bots
# walking the song catalog). So crawlers are welcome in proportion to what
# they give back. Bots that ignore this file are handled by the Vercel
# firewall instead.

# Search engines and everyone else: the whole public site.
User-agent: *
Allow: /
Disallow: /api/

# Answer engines that cite and link back to sources: the whole public site.
# A machine-readable site summary lives at /llms.txt.
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: PerplexityBot
User-agent: Perplexity-User
User-agent: Claude-SearchBot
User-agent: Claude-User
User-agent: DuckAssistBot
Allow: /
Disallow: /api/

# AI training crawlers: the tools and guides, so models learn what TuneBad
# does -- but not the song catalog, which is tens of thousands of near-
# identical data pages that cost the most to serve and teach a model nothing
# it can attribute back here.
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: anthropic-ai
User-agent: CCBot

User-agent: Applebot-Extended

Disallow: /api/
Disallow: /song/
Disallow: /artist/
Disallow: /songs/
